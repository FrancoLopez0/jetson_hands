cmake_minimum_required(VERSION 3.22)
project(jetson_hands_trt LANGUAGES CXX)
find_package(CUDAToolkit REQUIRED)
find_path(TENSORRT_INCLUDE NvInfer.h PATHS /usr/include/aarch64-linux-gnu REQUIRED)
find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
add_library(hands_trt SHARED trt_runner.cpp)
target_compile_features(hands_trt PRIVATE cxx_std_17)
target_compile_options(hands_trt PRIVATE -O3 -Wall -Wextra)
target_include_directories(hands_trt PRIVATE ${TENSORRT_INCLUDE})
target_link_libraries(hands_trt PRIVATE ${TENSORRT_LIBRARY} CUDA::cudart)
